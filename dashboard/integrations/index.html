Интеграция с Яндекс Картами, 2ГИС и другими сервисами

Сервис публикует положение ТС и прогноз прибытия, рассчитанные алгоритмами диспетчерской (ETA-движок + ML-прогноз задержки), в открытых стандартах. Картографический сервис подключает фиды — и автобусы на его картах двигаются и «опаздывают» так, как прогнозирует наша система.

1. Фиды GTFS-Realtime (рекомендуемый путь)

GTFS / GTFS-Realtime — стандарт, в котором перевозчики передают расписания и онлайн-данные агрегаторам (Яндекс Карты, 2ГИС, Google Maps, Moovit и др. принимают фиды через партнёрские программы для перевозчиков).

ФидURLСодержимое
Статическое расписание/api/export/gtfs-static.zipstops, routes, trips, stop_times, shapes (геометрия по дорогам)
VehiclePositions/api/export/gtfs-rt/vehicle-positionsкоординаты, курс, скорость; при потере связи — расчётное положение
TripUpdates/api/export/gtfs-rt/trip-updatesпрогноз прибытия на остановки с неопределённостью; на горизонте 10–15 мин — ML
Alerts/api/export/gtfs-rt/alertsинциденты: причина, эффект, маршрут

По умолчанию — protobuf (application/x-protobuf), для отладки — ?format=json. Идентификаторы stop_id / route_id / trip_id в RT совпадают со статическим GTFS.

3. Яндекс Карты (JavaScript API 3.0)

Требуется ключ JS API. Слой ТС строится из того же GeoJSON:

<script src="https://api-maps.yandex.ru/v3/?apikey=ВАШ_КЛЮЧ&lang=ru_RU"></script>
<script>
(async () => {
  await ymaps3.ready;
  const {YMap, YMapDefaultSchemeLayer, YMapDefaultFeaturesLayer, YMapMarker} = ymaps3;
  const map = new YMap(document.getElementById("map"), {location: {center: [37.62, 55.75], zoom: 11}});
  map.addChild(new YMapDefaultSchemeLayer()); map.addChild(new YMapDefaultFeaturesLayer());
  const markers = new Map();
  async function tick() {
    const fc = await (await fetch("https://ВАШ_ХОСТ/api/export/vehicles.geojson")).json();
    for (const f of fc.features) {
      const id = f.properties.unit_id, color = {red: "#d03b3b", yellow: "#fab219", green: "#0ca30c"}[f.properties.level] || "#898781";
      let m = markers.get(id);
      if (!m) { const el = document.createElement("div"); el.style.cssText = "width:14px;height:14px;border-radius:50%;border:2px solid #fff";
                m = new YMapMarker({coordinates: f.geometry.coordinates}, el); map.addChild(m); markers.set(id, m); }
      m.update({coordinates: f.geometry.coordinates}); m.element.style.background = color;
    }
  }
  tick(); setInterval(tick, 5000);
})();
</script>

4. 2ГИС (MapGL JS API)

Требуется ключ MapGL:

<script src="https://mapgl.2gis.com/api/js/v1"></script>
<script>
const map = new mapgl.Map("map", {center: [37.62, 55.75], zoom: 11, key: "ВАШ_КЛЮЧ"});
const markers = new Map();
async function tick() {
  const fc = await (await fetch("https://ВАШ_ХОСТ/api/export/vehicles.geojson")).json();
  for (const f of fc.features) {
    const id = f.properties.unit_id;
    markers.get(id)?.destroy();
    markers.set(id, new mapgl.CircleMarker(map, {coordinates: f.geometry.coordinates, radius: 8,
      color: {red: "#d03b3b", yellow: "#fab219", green: "#0ca30c"}[f.properties.level] || "#898781",
      strokeWidth: 2, strokeColor: "#fff"}));
  }
}
tick(); setInterval(tick, 5000);
</script>

CORS для /api/* открыт; в продакшене доступ к фидам ограничивается ключом партнёра на балансировщике.
